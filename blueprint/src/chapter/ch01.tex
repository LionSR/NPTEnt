\chapter{Bound entanglement with negative partial transpose}

A central open problem in quantum entanglement theory is whether every bipartite state with negative partial transpose (NPT) can be distilled into maximally entangled pairs by local operations and classical communication (LOCC), or whether NPT bound entangled states exist. Following~\cite{Faisal2026}, we develop a general single-copy operator certificate bounding all-copy two-qubit Bell overlap under separable operations, and apply it to the two-qudit Werner state $\rho = (23I - 3F)/1448$ on $\mathbb{C}^8\otimes\mathbb{C}^8$.

All Hilbert spaces below are finite-dimensional over $\mathbb{C}$; we write $\mathcal{L}(\mathcal{H})$ for linear operators on $\mathcal{H}$, $X^{\dagger}$ for the adjoint, and $X^T$ for the transpose in a fixed computational basis. The general operator-certificate framework, the explicit single-copy certificate for $\rho$, the uniform canonical Bell-overlap cap, zero finite-round, finite-outcome LOCC distillation rate $E_D(\rho)=0$, and finite-copy undistillability in the all-copy Schmidt-rank-$\le 2$ partial-transpose criterion are proved below. Infinite-round or closure LOCC extensions and the stronger arbitrary-filter trace-norm coherence bound are not claimed here.

\section{Symmetric order intervals and reverse-block positivity}

\begin{lemma}[Tensor stability of order intervals]
\label{lem:tensor-interval}
\lean{Matrix.piKronecker_add_sub_posSemidef}
\leanok
Let $n\ge 0$, and for each $k\in\{1,\dots,n\}$ let $A_k, B_k\in\mathcal{L}(\mathcal{H}_k)$ satisfy
\[
  A_k + B_k \succeq 0 \qquad\text{and}\qquad A_k - B_k \succeq 0.
\]
Then $A = \bigotimes_{k=1}^n A_k$ and $B = \bigotimes_{k=1}^n B_k$ satisfy $A + B \succeq 0$ and $A - B \succeq 0$ (with the empty product for $n=0$ equal to $1$ on $\mathbb{C}$).
\end{lemma}
\begin{proof}
\leanok
Induct on $n$. For $n=0$, $A=B=1$, so $A+B=2\succeq 0$ and $A-B=0\succeq 0$. The inductive step follows from the identities
\[
  A_1\otimes A_2 \pm B_1\otimes B_2
  = \tfrac{1}{2}\big((A_1+B_1)\otimes(A_2\pm B_2) + (A_1-B_1)\otimes(A_2\mp B_2)\big),
\]
since Kronecker products and sums preserve positive semidefiniteness.
\end{proof}

Applied to Choi matrices, this yields the same tensor stability for completely positive maps, i.e., linear maps $\Lambda : \mathcal{L}(\mathcal{H}_{\mathrm{in}}) \to \mathcal{L}(\mathcal{H}_{\mathrm{out}})$ such that $\mathrm{id}_k\otimes\Lambda$ preserves positive semidefiniteness for all $k\ge 1$.

\begin{lemma}[Off-diagonal agreement]
\label{lem:cp-agreement}
\lean{IsKrausCP.map_vecMulVec_right_eq_zero, IsKrausCP.map_vecMulVec_left_eq_zero}
\leanok
Let $\mathcal{D} : \mathcal{L}(\mathcal{H}_{\mathrm{in}}) \to \mathcal{L}(\mathcal{H}_{\mathrm{out}})$ be completely positive and $\ket{u}\in\mathcal{H}_{\mathrm{in}}$. If $\mathcal{D}(\ketbra{u}{u}) = 0$, then
\[
  \mathcal{D}(\ketbra{z}{u}) = 0 \qquad\text{and}\qquad \mathcal{D}(\ketbra{u}{z}) = 0
\]
for every $\ket{z}\in\mathcal{H}_{\mathrm{in}}$.
\end{lemma}
\begin{proof}
\leanok
Writing $\mathcal{D}(X) = \sum_i K_i X K_i^{\dagger}$, the hypothesis $\sum_i \ketbra{K_i u}{K_i u} = 0$ forces $\sum_i \|K_i u\|^2 = 0$, hence $K_i u = 0$ for every $i$. Therefore $\mathcal{D}(\ketbra{z}{u}) = \sum_i \ketbra{K_i z}{K_i u} = 0$ and $\mathcal{D}(\ketbra{u}{z}) = \sum_i \ketbra{K_i u}{K_i z} = 0$.
\end{proof}

For a linear map $\Lambda : \mathcal{L}(\mathcal{H}_{\mathrm{in}}) \to \mathcal{L}(\mathcal{H}_{\mathrm{out}})$, its (unnormalized, input-first) \emph{Choi matrix} in the computational basis $\{\ket{a}\}$ of $\mathcal{H}_{\mathrm{in}}$ is
\[
  J_\Lambda = \sum_{a,b} \ketbra{a}{b}\otimes\Lambda(\ketbra{a}{b})
  \in \mathcal{L}(\mathcal{H}_{\mathrm{in}}\otimes\mathcal{H}_{\mathrm{out}}).
\]
Then $\Lambda$ is completely positive if and only if $J_\Lambda\succeq 0$, preserves Hermiticity if and only if $J_\Lambda$ is Hermitian, and is \emph{completely copositive} ($\Lambda\circ T$ completely positive) if and only if $J_\Lambda^{\Gamma_{\mathrm{in}}}\succeq 0$, where $\Gamma_{\mathrm{in}}$ transposes $\mathcal{H}_{\mathrm{in}}$. For $\ket{u} = \sum_a u_a\ket{a}$, write $\ket{\bar{u}} = \sum_a \overline{u_a}\ket{a}$.

\begin{lemma}[Choi certificate]
\label{lem:choi-cert}
\lean{operatorSchmidtMap_satisfies_reverseBlock_hypotheses, exists_map_of_choiCertificate, exists_map_of_choiCertificate_compression}
\leanok
\uses{lem:cp-agreement}
Let $\mathcal{C}, \mathcal{S} : \mathcal{L}(\mathcal{H}_{\mathrm{in}}) \to \mathcal{L}(\mathcal{H}_{\mathrm{out}})$ be linear maps with Choi matrices $J = J_{\mathcal{C}}$ and $Q = J_{\mathcal{S}}$, and let $\ket{u}\in\mathcal{H}_{\mathrm{in}}$. The following are equivalent:
\begin{enumerate}
  \item $\mathcal{C}+\mathcal{S}$ and $\mathcal{C}-\mathcal{S}$ are completely positive, $\mathcal{S}$ is completely copositive, and $\mathcal{C}(\ketbra{u}{u}) = \mathcal{S}(\ketbra{u}{u})$ (equivalently, $\mathcal{C}$ and $\mathcal{S}$ agree on $\ketbra{z}{u}$ and $\ketbra{u}{z}$ for all $\ket{z}\in\mathcal{H}_{\mathrm{in}}$);
  \item $J+Q\succeq 0$, $J-Q\succeq 0$, $Q^{\Gamma_{\mathrm{in}}}\succeq 0$, and $(J-Q)(\ket{\bar{u}}\otimes I_{\mathrm{out}}) = 0$ (equivalently, $(\bra{\bar{u}}\otimes I_{\mathrm{out}})(J-Q)(\ket{\bar{u}}\otimes I_{\mathrm{out}}) = 0$).
\end{enumerate}
\end{lemma}
\begin{proof}
\leanok
Complete positivity of $\mathcal{C}\pm\mathcal{S}$ and complete copositivity of $\mathcal{S}$ correspond via the Choi isomorphism to $J\pm Q\succeq 0$ and $Q^{\Gamma_{\mathrm{in}}}\succeq 0$. Since $(\bra{\bar{u}}\otimes I_{\mathrm{out}})(J-Q)(\ket{\bar{u}}\otimes I_{\mathrm{out}}) = (\mathcal{C}-\mathcal{S})(\ketbra{u}{u})$ and $J-Q\succeq 0$, this compression vanishes if and only if $(J-Q)(\ket{\bar{u}}\otimes I_{\mathrm{out}}) = 0$; Lemma~\ref{lem:cp-agreement} gives the off-diagonal agreement.
\end{proof}

\begin{theorem}[Reverse-block positivity]
\label{thm:reverse-criterion}
\lean{piTensorMap_const_reverseBlock_posSemidef, piTensorMap_reverseBlock_posSemidef}
\leanok
\uses{lem:tensor-interval,lem:cp-agreement}
Let $\mathcal{C}, \mathcal{S} : \mathcal{L}(\mathcal{H}_{\mathrm{in}}) \to \mathcal{L}(\mathcal{H}_{\mathrm{out}})$ and $\ket{u}\in\mathcal{H}_{\mathrm{in}}$ be such that $\mathcal{C}+\mathcal{S}$ and $\mathcal{C}-\mathcal{S}$ are completely positive, $\mathcal{S}$ is completely copositive, and $\mathcal{C}(\ketbra{u}{u})=\mathcal{S}(\ketbra{u}{u})$. Then for every $n\ge 1$ and all $\ket{v},\ket{w}\in\mathcal{H}_{\mathrm{in}}^{\otimes n}$ with $\ket{v}+\ket{w}=\ket{u}^{\otimes n}$,
\[
  \Omega_{v,w} =
  \begin{pmatrix}
    \mathcal{C}^{\otimes n}(\ketbra{w}{w}) & \mathcal{C}^{\otimes n}(\ketbra{v}{w}) \\
    \mathcal{C}^{\otimes n}(\ketbra{w}{v}) & \mathcal{C}^{\otimes n}(\ketbra{v}{v})
  \end{pmatrix}
  \succeq 0.
\]
The same holds for any finite heterogeneous tensor product $\bigotimes_{k=1}^n \mathcal{C}_k$ and $\ket{u} = \bigotimes_{k=1}^n \ket{u^{(k)}}$ whose factors each satisfy these hypotheses.
\end{theorem}
\begin{proof}
\leanok
Write $\mathcal{C}_n = \bigotimes_{k=1}^n \mathcal{C}_k$, $\mathcal{S}_n = \bigotimes_{k=1}^n \mathcal{S}_k$, and $\ket{u_n} = \bigotimes_{k=1}^n \ket{u^{(k)}}$. By Lemma~\ref{lem:tensor-interval} on Choi matrices, $\mathcal{C}_n - \mathcal{S}_n$ is completely positive, while $\mathcal{S}_n$ is completely copositive and agrees with $\mathcal{C}_n$ on $\ketbra{u_n}{u_n}$, hence also on $\ketbra{v}{u_n}$ and $\ketbra{u_n}{v}$ by Lemma~\ref{lem:cp-agreement}. Adding $\operatorname{diag}(0, (\mathcal{C}_n-\mathcal{S}_n)(\ketbra{v}{v}))\succeq 0$ to the positive semidefinite reverse-block matrix of $\mathcal{S}_n$ on $(\ket{u_n},\ket{v})$ gives $\bigl(\begin{smallmatrix}\mathcal{C}_n(\ketbra{u_n}{u_n}) & \mathcal{C}_n(\ketbra{v}{u_n}) \\ \mathcal{C}_n(\ketbra{u_n}{v}) & \mathcal{C}_n(\ketbra{v}{v})\end{smallmatrix}\bigr)\succeq 0$. Conjugating by $\bigl(\begin{smallmatrix}I & -I \\ 0 & I\end{smallmatrix}\bigr)$ and substituting $\ket{w}=\ket{u_n}-\ket{v}$ yields $\Omega_{v,w}\succeq 0$.
\end{proof}

\section{Local factorization and Bell-overlap obstruction}

\begin{lemma}[Spectral splitting]
\label{lem:spectral-split}
\lean{Matrix.exists_twoForms_bipartite_split}
\leanok
Let $E_0, E_1 \in \mathcal{L}(\mathcal{H}_A)$ be positive semidefinite, let $\ket{u}\in\mathcal{H}_A\otimes\mathcal{H}_B$, and let $\{F_j\}_{j\in J}$ be a family of positive semidefinite operators on $\mathcal{H}_B$. Then there exist $\ket{v},\ket{w}\in\mathcal{H}_A\otimes\mathcal{H}_B$ with $\ket{v}+\ket{w}=\ket{u}$ such that for every $j\in J$ and $i\in\{0,1\}$,
\[
  \bra{w}(E_i\otimes F_j)\ket{v} = \bra{v}(E_i\otimes F_j)\ket{w} = 0, \qquad
  \bra{v}\big((E_0-E_1)\otimes F_j\big)\ket{v} \ge 0, \qquad
  \bra{w}\big((E_1-E_0)\otimes F_j\big)\ket{w} \ge 0.
\]
\end{lemma}
\begin{proof}
\leanok
Let $H = E_0 + E_1 \succeq 0$, let $W$ be the invertible Hermitian operator equal to $H^{-1/2}$ on $\operatorname{supp}(H)$ and $I$ on $\ker(H)$, and let $P_+$ be the spectral projection onto the nonnegative eigenspace of $W(E_0 - E_1)W$. Setting $Q = W P_+ W^{-1}$, the operators $Q\otimes I_B$ and $(I_A - Q)\otimes I_B$ split $\ket{u} = \ket{v} + \ket{w}$ so that $E_0, E_1$ have vanishing cross-terms and $E_0 - E_1$ is positive semidefinite on the $Q$-component and negative semidefinite on the $(I_A - Q)$-component; tensoring with $F_j\succeq 0$ preserves both properties.
\end{proof}

For a product of completely positive maps $\mathcal{C} = L\otimes M$ and positive semidefinite local effects $X_0, X_1$ and $Y_j$, applying Lemma~\ref{lem:spectral-split} to $E_i = L^{\dagger}(X_i)$ and $F_j = M^{\dagger}(Y_j)$ splits $\ket{u}$ with respect to the adjoint local effects. For $n$ copies of a bipartite space $\mathcal{H}_A\otimes\mathcal{H}_B$, we pass between the copy-major ordering $(\mathcal{H}_A\otimes\mathcal{H}_B)^{\otimes n}$ and the party-major ordering $\mathcal{H}_A^{\otimes n}\otimes\mathcal{H}_B^{\otimes n}$ by the canonical factor permutation grouping Alice's systems before Bob's.

\begin{lemma}[Rank-one coherence bound]
\label{lem:coherence-general}
\lean{QuantumInfo.rankOne_coherence_pow_four_le, QuantumInfo.rankOne_coherence_pow_four_le_of_smul}
\leanok
\uses{thm:reverse-criterion,lem:spectral-split}
Let $L : \mathcal{L}(\mathcal{H}_{A,\mathrm{in}}) \to \mathcal{L}(\mathcal{H}_{A,\mathrm{out}})$ and $M : \mathcal{L}(\mathcal{H}_{B,\mathrm{in}}) \to \mathcal{L}(\mathcal{H}_{B,\mathrm{out}})$ be completely positive, $\mathcal{C} = L\otimes M$, and $\ket{u}\in\mathcal{H}_{A,\mathrm{in}}\otimes\mathcal{H}_{B,\mathrm{in}}$ with $\Tr(\mathcal{C}(\ketbra{u}{u})) > 0$. Set $\rho = \mathcal{C}(\ketbra{u}{u}) / \Tr(\mathcal{C}(\ketbra{u}{u}))$, and suppose there exists $\mathcal{S}$ satisfying the hypotheses of Theorem~\ref{thm:reverse-criterion} for $(\mathcal{C}, \ket{u})$. Then for every $n\ge 1$ and all $\ket{a_0},\ket{a_1}\in\mathcal{H}_{A,\mathrm{out}}^{\otimes n}$, $\ket{b_0},\ket{b_1}\in\mathcal{H}_{B,\mathrm{out}}^{\otimes n}$ (in party-major coordinates), with $\ket{x_{ij}} = \ket{a_i}\otimes\ket{b_j}$, $p_{ij} = \bra{x_{ij}}\rho^{\otimes n}\ket{x_{ij}}$, and $k = |\bra{x_{00}}\rho^{\otimes n}\ket{x_{11}}|$,
\[
  k^4 \le p_{00}\,p_{01}\,p_{10}\,p_{11}.
\]
\end{lemma}
\begin{proof}
\leanok
Applying Lemma~\ref{lem:spectral-split} to the adjoint local effects $(L^{\otimes n})^{\dagger}(\ketbra{a_i}{a_i})$ and $(M^{\otimes n})^{\dagger}(\ketbra{b_j}{b_j})$ splits $\ket{u}^{\otimes n} = \ket{v} + \ket{w}$. Compressing $\Omega_{v,w}\succeq 0$ from Theorem~\ref{thm:reverse-criterion} against $(\ket{x_{00}}, \ket{x_{11}})$ and using positivity of the diagonal blocks bounds $k$ by a four-term square-root sum in the split populations, whose optimization under the splitting inequalities yields $k^4 \le p_{00}p_{01}p_{10}p_{11}$.
\end{proof}

Fix the canonical two-qubit Bell state $\ket{\Omega_2} = (\ket{00}+\ket{11})/\sqrt{2}$ with projector $\Omega_2 = \ketbra{\Omega_2}{\Omega_2}$.

\begin{theorem}[All-copy Bell-overlap bound]
\label{thm:overlap-criterion}
\lean{QuantumInfo.bellOverlap_separableOperation_le, QuantumInfo.bellOverlap_localUnitary_separableOperation_le, QuantumInfo.bellOverlap_normalized_separableOperation_le}
\leanok
\uses{def:sep-op,lem:coherence-general}
Under the hypotheses of Lemma~\ref{lem:coherence-general}, for every $n\ge 1$, every product-Kraus separable operation $\Lambda$ from $\mathcal{H}_{A,\mathrm{out}}^{\otimes n}\otimes\mathcal{H}_{B,\mathrm{out}}^{\otimes n}$ to $\mathbb{C}^2\otimes\mathbb{C}^2$ (Definition~\ref{def:sep-op}), and every local-unitary equivalent $\Phi = (U\otimes V)\Omega_2(U\otimes V)^{\dagger}$,
\[
  \Tr\big(\Phi\,\Lambda(\rho^{\otimes n})\big)
  \le \frac{1+\sqrt{2}}{4}\,\Tr\big(\Lambda(\rho^{\otimes n})\big).
\]
Hence, whenever $\Tr(\Lambda(\rho^{\otimes n})) > 0$, the normalized output $\tau = \Lambda(\rho^{\otimes n})/\Tr(\Lambda(\rho^{\otimes n}))$ satisfies $\Tr(\Omega_2\tau) \le (1+\sqrt{2})/4$.
\end{theorem}
\begin{proof}
\leanok
Absorbing $U^{\dagger}\otimes V^{\dagger}$ into the local Kraus operators reduces to $\Phi = \Omega_2$. For each product Kraus term $A_\alpha\otimes B_\alpha$, applying Lemma~\ref{lem:coherence-general} to the rows of $A_\alpha$ and $B_\alpha$ gives $2|\bra{00}\tau_\alpha\ket{11}| \le 2(p_{00}p_{01}p_{10}p_{11})^{1/4}$, whence $\Tr(\Omega_2\tau_\alpha) \le \frac{1+\sqrt{2}}{4}\Tr(\tau_\alpha)$ by AM--GM; summing over $\alpha$ concludes the proof.
\end{proof}

\section{Separable operations and zero distillation}

\begin{definition}[Product-Kraus separable operation]
\label{def:sep-op}
\lean{QuantumInfo.SeparableOperation, QuantumInfo.SeparableOperation.map, QuantumInfo.SeparableOperation.IsTP, QuantumInfo.SeparableOperation.IsTraceNonincreasing}
\leanok
A \emph{product-Kraus separable operation} $S$ from $\mathbb{C}^{d_A}\otimes\mathbb{C}^{d_B}$ to $\mathbb{C}^{d_{A'}}\otimes\mathbb{C}^{d_{B'}}$ consists of a finite index set $I$ (possibly empty) and local operators $A_\alpha : \mathbb{C}^{d_A}\to\mathbb{C}^{d_{A'}}$, $B_\alpha : \mathbb{C}^{d_B}\to\mathbb{C}^{d_{B'}}$ ($\alpha\in I$), inducing the linear map
\[
  \Lambda_S(X) = \sum_{\alpha\in I} (A_\alpha\otimes B_\alpha)\,X\,(A_\alpha\otimes B_\alpha)^{\dagger}.
\]
It is \emph{trace-nonincreasing} if $\sum_{\alpha\in I} (A_\alpha^{\dagger}A_\alpha)\otimes(B_\alpha^{\dagger}B_\alpha) \preceq I$, and \emph{trace-preserving} if equality holds.
\end{definition}

\begin{lemma}[Separable-operation calculus]
\label{lem:sep-op-comp}
\lean{QuantumInfo.SeparableOperation.isKrausCP_map, QuantumInfo.SeparableOperation.map_id, QuantumInfo.SeparableOperation.comp, QuantumInfo.SeparableOperation.map_comp, QuantumInfo.SeparableOperation.map_comp_eq, KrausFamily.trace_map, KrausFamily.trace_map_le}
\leanok
\uses{def:sep-op}
For every product-Kraus separable operation $S$, $\Lambda_S$ is completely positive. The singleton pair $(I_A, I_B)$ induces the identity map, and composable operations $S, T$ compose to $T\circ S$ indexed by $I_T\times I_S$ with local operators $(A'_\beta A_\alpha, B'_\beta B_\alpha)$ and $\Lambda_{T\circ S} = \Lambda_T\circ\Lambda_S$. Moreover, trace-preserving operations satisfy $\Tr(\Lambda_S(X))=\Tr(X)$ for all $X$, and trace-nonincreasing operations satisfy $\Tr(\Lambda_S(\sigma))\le\Tr(\sigma)$ for all $\sigma\succeq 0$.
\end{lemma}
\begin{proof}
\leanok
Viewing $(A_\alpha\otimes B_\alpha)_{\alpha\in I}$ as a Kraus family gives complete positivity and both trace relations directly. Composition follows from $(A'_\beta\otimes B'_\beta)(A_\alpha\otimes B_\alpha) = (A'_\beta A_\alpha)\otimes(B'_\beta B_\alpha)$ and reindexing over $I_T\times I_S$.
\end{proof}

A \emph{state} (density matrix) is a positive semidefinite operator $\sigma\succeq 0$ with $\Tr(\sigma)=1$. Measure error by the full trace norm $\|X\|_1 = \Tr\sqrt{X^{\dagger}X}$, with target $\Omega_2$. Given a bipartite state $\sigma$ and a class $\mathcal{P}$ of trace-preserving bipartite protocols mapping $n$ party-major copies to $m$ qubit pairs, a rate $r\ge 0$ is \emph{achievable} if there exist protocols $\Lambda_n\in\mathcal{P}(n,\lfloor rn\rfloor)$ for all sufficiently large $n$ with $\|\Lambda_n(\sigma^{\otimes n}) - \Omega_2^{\otimes\lfloor rn\rfloor}\|_1 \to 0$. Equivalently, defining in $[0,+\infty]$ for $r\ge 0$ and $n\ge 1$
\[
  \varepsilon_n^{\mathcal{P}}(r) = \inf_{\Lambda\in\mathcal{P}(n,\lfloor rn\rfloor)}
  \big\|\Lambda(\sigma^{\otimes n}) - \Omega_2^{\otimes\lfloor rn\rfloor}\big\|_1,
  \qquad
  E_D^{\mathcal{P}}(\sigma) = \sup\big\{r\ge 0 : \lim_{n\to\infty}\varepsilon_n^{\mathcal{P}}(r)=0\big\},
\]
with $\inf\emptyset = +\infty$ and $\sup\emptyset = 0$, $r$ is achievable if and only if $\lim_{n\to\infty}\varepsilon_n^{\mathcal{P}}(r)=0$. An entangled state $\sigma$ is \emph{bound entangled} under $\mathcal{P}$ if $E_D^{\mathcal{P}}(\sigma)=0$.

\begin{lemma}[Zero-distillation criterion]
\label{lem:zero}
\lean{QuantumCircuit.distillableEntanglement_eq_zero_of_uniformBellOverlapCap, QuantumCircuit.isAchievableDistillationRate_iff_tendsto_minDistillationError}
\leanok
\uses{def:sep-op,lem:sep-op-comp}
Let $\sigma$ be a bipartite state and $\mathcal{P}$ a class of trace-preserving bipartite protocols whose maps are all induced by product-Kraus separable operations. If there exists $c<1$ such that
\[
  \Tr\big(\Omega_2\,\Lambda(\sigma^{\otimes n})\big)
  \le c\,\Tr\big(\Lambda(\sigma^{\otimes n})\big)
\]
for every $n\ge 1$ and every product-Kraus separable operation $\Lambda$ mapping $n$ copies to $\mathbb{C}^2\otimes\mathbb{C}^2$, then $E_D^{\mathcal{P}}(\sigma)=0$.
\end{lemma}
\begin{proof}
\leanok
If $r>0$ were achievable (equivalently $\varepsilon_n^{\mathcal{P}}(r)\to 0$), then for sufficiently large $n$ we have $\lfloor rn\rfloor\ge 1$ and some $\Lambda_n\in\mathcal{P}(n,\lfloor rn\rfloor)$ with $\|\Lambda_n(\sigma^{\otimes n}) - \Omega_2^{\otimes\lfloor rn\rfloor}\|_1 < 1-c$. Composing $\Lambda_n$ with local discard of all output pairs except the first---a trace-preserving product-Kraus separable operation that contracts $\|\cdot\|_1$ and sends $\Omega_2^{\otimes\lfloor rn\rfloor}$ to $\Omega_2$---produces a unit-trace two-qubit state $\tau_n$ with $\|\tau_n - \Omega_2\|_1 < 1-c$, hence $\Tr(\Omega_2\tau_n) > c$, contradicting the uniform cap.
\end{proof}

\begin{lemma}[Finite-round LOCC is separable]
\label{lem:locc}
\lean{QuantumCircuit.IsFiniteRoundLOCC.exists_separableOperation}
\leanok
\uses{def:sep-op,lem:sep-op-comp}
Every finite-round, finite-outcome adaptive bipartite protocol built from outcome-dependent local quantum instruments with classical feed-forward has product Kraus operators on each outcome branch, and its averaged trace-preserving map is a product-Kraus separable operation.
\end{lemma}
\begin{proof}
\leanok
Induct on the finite adaptive round tree. At each round, the parallel local trace-preserving Kraus instruments have joint Kraus operators factoring across the two parties, and composing with the inductively constructed trace-preserving separable operations on each outcome branch preserves both the product-Kraus form and trace preservation.
\end{proof}

\section{Finite-copy undistillability and recurrence amplification}

Separately from the finite-round LOCC distillation rate $E_D^{\mathcal{P}}(\sigma)$ of Lemma~\ref{lem:zero}, we consider the finite-copy distillability criterion~\cite{Lewenstein2000Primer} expressed via Schmidt-rank-$\le 2$ witnesses of the partial transpose and local $2\times 2$ projections/filters on $K$ party-major copies.

\begin{definition}[Finite-copy undistillability]
\label{def:finite-copy-undistillable}
\lean{QuantumInfo.IsNDistillable, QuantumInfo.IsNUndistillable, QuantumInfo.IsFiniteCopyDistillable, QuantumInfo.IsFiniteCopyUndistillable, QuantumInfo.isFiniteCopyUndistillable_of_partialTransposeFst_posSemidef}
\leanok
For a bipartite operator $\sigma$ on $\mathcal{H}_A\otimes\mathcal{H}_B$ and $K\ge 1$, write $(\sigma^{\otimes K})^{\Gamma_A}$ for the partial transpose on Alice's $K$-copy subsystem (in party-major coordinates). We call $\sigma$ \emph{$K$-distillable} if $K>0$ and there exists a vector $\ket{\psi}\in\mathcal{H}_A^{\otimes K}\otimes\mathcal{H}_B^{\otimes K}$ of Schmidt rank at most $2$ such that $\operatorname{Re}\bra{\psi}(\sigma^{\otimes K})^{\Gamma_A}\ket{\psi} < 0$, and \emph{$K$-undistillable} otherwise. We call $\sigma$ \emph{finite-copy distillable} if it is $K$-distillable for some $K\ge 1$, and \emph{finite-copy undistillable} if it is $K$-undistillable for every $K\ge 1$. In particular, every PPT operator ($\sigma^{\Gamma_A}\succeq 0$) is finite-copy undistillable.
\end{definition}

\begin{lemma}[Two-qubit projection and local filtering criterion]
\label{lem:two-qubit-filter}
\lean{QuantumInfo.isNDistillable_iff_dclbTwoTermForm, QuantumInfo.isNDistillable_iff_exists_coisometry_twoQubit_nppt, QuantumInfo.isNDistillable_iff_exists_filter_twoQubit_nppt, NPTEnt.TwoQubitDistillability.not_isPPT_twoQubit_iff_exists_localFilter_normalized_bellOverlap_gt_half, NPTEnt.TwoQubitDistillability.isNDistillable_iff_exists_localFilter_normalized_bellOverlap_gt_half, NPTEnt.TwoQubitDistillability.isFiniteCopyDistillable_iff_exists_localFilter_normalized_bellOverlap_gt_half}
\leanok
\uses{def:finite-copy-undistillable}
Let $\sigma\succeq 0$ on $\mathcal{H}_A\otimes\mathcal{H}_B$ and $K\ge 1$. The following are equivalent:
\begin{enumerate}
  \item $\sigma$ is $K$-distillable;
  \item there exist orthonormal local pairs $\{\ket{e_1},\ket{e_2}\}\subset\mathcal{H}_A^{\otimes K}$ and $\{\ket{f_1},\ket{f_2}\}\subset\mathcal{H}_B^{\otimes K}$ and scalars $a,b\in\mathbb{C}$ such that $\ket{\psi} = a\ket{e_1^*}\ket{f_1} + b\ket{e_2^*}\ket{f_2}$ (where $\ket{e_i^*}$ denotes entrywise complex conjugation) satisfies $\operatorname{Re}\bra{\psi}(\sigma^{\otimes K})^{\Gamma_A}\ket{\psi} < 0$;
  \item there exist local $2$-dimensional projections/filters $K_A : \mathcal{H}_A^{\otimes K}\to\mathbb{C}^2$ and $K_B : \mathcal{H}_B^{\otimes K}\to\mathbb{C}^2$ such that $(K_A\otimes K_B)\sigma^{\otimes K}(K_A\otimes K_B)^{\dagger}$ is NPT;
  \item there exist local filters $K_A : \mathcal{H}_A^{\otimes K}\to\mathbb{C}^2$ and $K_B : \mathcal{H}_B^{\otimes K}\to\mathbb{C}^2$ such that $\tau = (K_A\otimes K_B)\sigma^{\otimes K}(K_A\otimes K_B)^{\dagger}$ satisfies $\Tr(\tau) > 0$ and
  \[
    \frac{\Tr(\Omega_2\,\tau)}{\Tr(\tau)} > \frac{1}{2}.
  \]
\end{enumerate}
\end{lemma}
\begin{proof}
\leanok
Since $\sigma\succeq 0$, any product vector $\ket{u}\ket{v}$ satisfies $\bra{u,v}(\sigma^{\otimes K})^{\Gamma_A}\ket{u,v} = \bra{u^*,v}\sigma^{\otimes K}\ket{u^*,v}\ge 0$. Hence any negative Schmidt-rank-$\le 2$ witness $\ket{\psi}$ has Schmidt rank $2$, forcing local dimensions at least $2$ and giving the orthonormal two-term representation $\ket{\psi} = a\ket{e_1^*}\ket{f_1} + b\ket{e_2^*}\ket{f_2}$, which yields $(1)\Leftrightarrow(2)$. Pulling back the Alice partial transpose along local $2$-dimensional coisometries/filters identifies such a negative witness with a two-qubit NPT output $\tau_0$, giving $(2)\Leftrightarrow(3)$. On $\mathbb{C}^2\otimes\mathbb{C}^2$, the identity $\operatorname{Re}\Tr(\tau) - 2\Tr(\Omega_2\tau) = \operatorname{Re}\bra{\psi^-}\tau^{\Gamma_A}\ket{\psi^-}$ for $\ket{\psi^-} = \ket{01} - \ket{10}$, combined with an explicit local filter pulling back a negative witness of $\tau_0^{\Gamma_A}$ to $\ket{\psi^-}$, yields a positive-trace two-qubit output with normalized Bell overlap strictly greater than $1/2$. Conversely, any PPT two-qubit operator satisfies $\Tr(\Omega_2\tau)\le \Tr(\tau)/2$, giving $(3)\Leftrightarrow(4)$.
\end{proof}

\begin{lemma}[Finite isotropic twirl and recurrence amplification]
\label{lem:recurrence-amplification}
\lean{NPTEnt.QubitRecurrence.isotropicTwirl12, NPTEnt.QubitRecurrence.isotropicTwirl12_map_of_posSemidef, NPTEnt.QubitRecurrence.recurrenceStep_map_isotropicState2, NPTEnt.RecurrenceAmplification.exists_iterate_qubitRecurrence_gt, NPTEnt.QubitRecurrence.exists_separableOperation_bellFraction_gt, NPTEnt.QubitRecurrence.exists_separableOperation_bellFraction_gt_of_filter}
\leanok
\uses{def:sep-op,lem:sep-op-comp}
Let $\tau\succeq 0$ on $\mathbb{C}^2\otimes\mathbb{C}^2$ satisfy $\Tr(\tau) > 0$ and $\Tr(\Omega_2\tau)/\Tr(\tau) > 1/2$. Then for every $c < 1$, there exist $M\ge 1$ and an unrestricted finite product-Kraus separable operation $\Lambda$ from $(\mathbb{C}^2)^{\otimes M}\otimes(\mathbb{C}^2)^{\otimes M}$ to $\mathbb{C}^2\otimes\mathbb{C}^2$ (not presumed trace-nonincreasing) such that $\Tr(\Lambda(\tau^{\otimes M})) > 0$ and
\[
  \frac{\Tr\big(\Omega_2\,\Lambda(\tau^{\otimes M})\big)}{\Tr\big(\Lambda(\tau^{\otimes M})\big)} > c.
\]
\end{lemma}
\begin{proof}
\leanok
Composing the $4$-element Pauli twirl with the $3$-element tetrahedral twirl yields a $12$-element product-Kraus separable operation $\mathcal{T}_{12}$ whose algebraic normalization is $12$ times the averaged isotropic twirl; on $\tau$ it outputs $12\Tr(\tau)\,\rho_f$, where $\rho_f = f\Omega_2 + \frac{1-f}{3}(I - \Omega_2)$ and $f = \Tr(\Omega_2\tau)/\Tr(\tau)$. Applying the bilateral even-parity filter $A\otimes A$ ($A\ket{ii}=\ket{i}$, $A\ket{ij}=0$ for $i\neq j$) to $\rho_f^{\otimes 2}$ followed by $\mathcal{T}_{12}$ produces $\frac{2(8f^2-4f+5)}{3}\,\rho_{g(f)}$, where
\[
  g(f) = \frac{10f^2 - 2f + 1}{8f^2 - 4f + 5}.
\]
On $(1/2, 1)$, the odds $(f - 1/2)/(1 - f)$ increase by at least $6/5$ at each step, while $g(1) = 1$. Iterating this step in a balanced binary tree on $M = 2^n$ copies yields a finite product-Kraus separable operation whose positive-trace output achieves normalized Bell overlap $g^{\circ n}(f) > c$ for sufficiently large $n$.
\end{proof}

\begin{theorem}[Uniform cap implies finite-copy undistillability]
\label{thm:cap-undistillable}
\lean{NPTEnt.TwoQubitDistillability.exists_separableOperation_bellFraction_gt_of_isNDistillable, NPTEnt.TwoQubitDistillability.isFiniteCopyUndistillable_of_uniform_bellOverlap_cap}
\leanok
\uses{def:sep-op,lem:sep-op-comp,def:finite-copy-undistillable,lem:two-qubit-filter,lem:recurrence-amplification}
Let $\sigma\succeq 0$ on $\mathcal{H}_A\otimes\mathcal{H}_B$. Suppose there exists $c < 1$ such that
\[
  \Tr\big(\Omega_2\,\Lambda(\sigma^{\otimes M})\big)
  \le c\,\Tr\big(\Lambda(\sigma^{\otimes M})\big)
\]
for every $M\ge 1$ and every finite product-Kraus separable operation $\Lambda$ from $\mathcal{H}_A^{\otimes M}\otimes\mathcal{H}_B^{\otimes M}$ to $\mathbb{C}^2\otimes\mathbb{C}^2$. Then $\sigma$ is finite-copy undistillable.
\end{theorem}
\begin{proof}
\leanok
Suppose $\sigma$ were $N$-distillable for some $N\ge 1$. By Lemma~\ref{lem:two-qubit-filter}, a single-term product-Kraus filter $S$ maps $\sigma^{\otimes N}$ to a positive-trace two-qubit operator $\tau\succeq 0$ with normalized Bell overlap exceeding $1/2$. Applying Lemma~\ref{lem:recurrence-amplification} with threshold $c$ yields $K\ge 1$ and a finite product-Kraus separable operation $\Lambda_0$ on $K$ copies of $\tau$ exceeding $c$. Tensoring $K$ copies of $S$ with explicit copy regrouping and composing with $\Lambda_0$ gives an unrestricted finite product-Kraus separable operation on $M = KN$ copies of $\sigma$ with positive output trace and normalized Bell overlap strictly exceeding $c$, contradicting the uniform homogeneous cap.
\end{proof}

\section{The two-qudit Werner state}

A bipartite state is \emph{separable} if it is a convex combination of product states $\sigma_A\otimes\sigma_B$, and \emph{entangled} otherwise. The \emph{partial transpose} $\sigma^{\Gamma}$ transposes the second factor, $(X\otimes Y)^{\Gamma}=X\otimes Y^T$; a state is \emph{PPT} if $\sigma^{\Gamma}\succeq 0$ and \emph{NPT} otherwise.

On $\mathbb{C}^8\otimes\mathbb{C}^8$ with computational basis $\ket{i,j}$ ($0\le i,j\le 7$), let $F\ket{i,j}=\ket{j,i}$ be the swap operator, let $\ket{u}=\sum_{i=0}^{7}\ket{i,i}$ (so $\braket{u}{u}=8$ and $\ket{\bar{u}}=\ket{u}$), and define
\[
  \rho = \frac{23I - 3F}{1448}.
\]

\begin{lemma}[State and NPT witness]
\label{lem:state}
\lean{NPTEnt.Werner.rho_state_npt, NPTEnt.Werner.partialTransposeRight_rho_mulVec_u}
\leanok
The operator $\rho$ is a density matrix with $\rho^{\Gamma}\ket{u}=-\ket{u}/1448$, and is therefore NPT.
\end{lemma}
\begin{proof}
\leanok
Since $F$ is Hermitian with eigenvalues $\pm 1$, $23I-3F = 20I+3(I-F)\succeq 0$. As $\Tr(I)=64$ and $\Tr(F)=8$, $\Tr(23I-3F)=1448$, so $\Tr(\rho)=1$. From $F=\sum_{i,j}\ketbra{j}{i}\otimes\ketbra{i}{j}$ we get $F^{\Gamma}=\ketbra{u}{u}$, hence $\rho^{\Gamma}=(23I-3\ketbra{u}{u})/1448$. Because $\braket{u}{u}=8$, $\rho^{\Gamma}\ket{u}=-\ket{u}/1448$ and $\bra{u}\rho^{\Gamma}\ket{u}=-8/1448<0$.
\end{proof}

\begin{lemma}[Entanglement of $\rho$]
\label{lem:nonseparable}
\lean{NPTEnt.Werner.rho_not_isSeparable}
\leanok
\uses{lem:state}
The state $\rho$ is entangled.
\end{lemma}
\begin{proof}
\leanok
Transposition preserves positive semidefiniteness, so every product state---and by linearity every separable state---is PPT. Since $\rho$ is NPT by Lemma~\ref{lem:state}, it is not separable.
\end{proof}

Define linear maps on $8\times 8$ matrices by
\[
  L(X) = 5\Tr(X)\,I - 3X^T, \qquad M(X) = 18\Tr(X)\,I + 37X,
\]
and set $\mathcal{C} = L\otimes M$, $\mathcal{C}_n = \mathcal{C}^{\otimes n}$, and $\ket{u_n} = \ket{u}^{\otimes n}$.

\begin{lemma}[Local-map representation]
\label{lem:maps}
\lean{NPTEnt.LocalMaps.L_isKrausCP, NPTEnt.LocalMaps.M_isKrausCP, NPTEnt.LocalMaps.piKronecker_rho_eq_smul_CCopies}
\leanok
The maps $L$ and $M$ are completely positive, and for every $n\ge 1$,
\[
  \rho^{\otimes n} = \frac{\mathcal{C}_n(\ketbra{u_n}{u_n})}{(37\cdot 1448)^n}.
\]
\end{lemma}
\begin{proof}
\leanok
Their Choi matrices $J_L = 5I - 3F \succeq 0$ and $J_M = 18I + 37\ketbra{u}{u} \succeq 0$ establish complete positivity. Direct evaluation gives $\mathcal{C}(\ketbra{u}{u}) = 37(23I - 3F) = 37\cdot 1448\,\rho$, and taking $n$-fold tensor products yields the $n$-copy identity.
\end{proof}

Let $J = J_{\mathcal{C}}$ be the Choi matrix of $\mathcal{C}$ on $(\mathbb{C}^8)^{\otimes 4}$ (ordered as Alice input, Bob input, Alice output, Bob output).

\begin{lemma}[Single-copy certificate]
\label{lem:certificate}
\lean{NPTEnt.Certificate.appendixA_certificate, NPTEnt.Main.auxiliaryMapS_spec}
\leanok
\uses{lem:maps}
There exists a linear map $\mathcal{S}$ on $64\times 64$ matrices whose Choi matrix $Q = J_{\mathcal{S}}$ is Hermitian and satisfies
\[
  J + Q \succeq 0, \qquad J - Q \succeq 0, \qquad
  Q^{\Gamma_{\mathrm{in}}} \succeq 0, \qquad
  (J - Q)(\ket{u}\otimes I) = 0,
\]
where $I$ is the identity on the two output systems.
\end{lemma}
\begin{proof}
\leanok
In Appendix~A of~\cite{Faisal2026}, $Q$ is given explicitly as a rational linear combination (denominator $40$) of partial transposes on the second factor of the $24$ permutation operators on $(\mathbb{C}^8)^{\otimes 4}$, commuting with $\bar{U}\otimes U\otimes\bar{U}\otimes\bar{U}$ for all $U\in U(8)$. Decomposing into canonical $U(8)$-invariant weight blocks reduces $J\pm Q\succeq 0$ and $Q^{\Gamma_{\mathrm{in}}}\succeq 0$ to exact rational $LDL^{\dagger}$ certificates on all canonical blocks, while $(J-Q)(\ket{u}\otimes I)=0$ holds directly; the inverse Choi map yields $\mathcal{S}$.
\end{proof}

Specializing Theorem~\ref{thm:reverse-criterion}, Lemma~\ref{lem:coherence-general}, and Theorem~\ref{thm:overlap-criterion} via Lemmas~\ref{lem:maps} and~\ref{lem:certificate} yields the following consequences for $\rho$.

\begin{lemma}[All-copy reverse-block positivity]
\label{lem:reverse}
\lean{NPTEnt.Main.CCopies_reverseBlock_posSemidef}
\leanok

For every $n\ge 0$ and all $\ket{v},\ket{w}\in(\mathbb{C}^8\otimes\mathbb{C}^8)^{\otimes n}$ (in copy-major coordinates, including the empty tensor power $n=0$) with $\ket{v}+\ket{w}=\ket{u_n}$,
\[
  \Omega_{v,w} =
  \begin{pmatrix}
    \mathcal{C}_n(\ketbra{w}{w}) & \mathcal{C}_n(\ketbra{v}{w}) \\
    \mathcal{C}_n(\ketbra{w}{v}) & \mathcal{C}_n(\ketbra{v}{v})
  \end{pmatrix}
  \succeq 0.
\]
\end{lemma}
\begin{proof}
\leanok
Combine Lemmas~\ref{lem:maps}, \ref{lem:certificate}, and~\ref{lem:choi-cert} with Theorem~\ref{thm:reverse-criterion}.

\begin{lemma}[All-copy coherence bound]
\label{lem:coherence}
\lean{NPTEnt.Main.rankOne_coherence_pow_four_le}
\leanok

For every $n\ge 0$ and all $\ket{a_0},\ket{a_1},\ket{b_0},\ket{b_1}\in(\mathbb{C}^8)^{\otimes n}$ (in party-major coordinates, without unit-norm restriction), with $\ket{x_{ij}} = \ket{a_i}\otimes\ket{b_j}$, $p_{ij} = \operatorname{Re}\bra{x_{ij}}\rho^{\otimes n}\ket{x_{ij}}$, and $k = |\bra{x_{00}}\rho^{\otimes n}\ket{x_{11}}|$,

\begin{theorem}[Uniform canonical Bell-overlap cap]
\label{thm:cap}
\lean{NPTEnt.Main.bellOverlap_separableOperation_le, NPTEnt.Main.bellOverlap_normalized_separableOperation_le, NPTEnt.Main.hasUniformBellOverlapCap_rho}
\leanok

For every $n\ge 1$ and every product-Kraus separable operation $\Lambda$ from $(\mathbb{C}^8)^{\otimes n}\otimes(\mathbb{C}^8)^{\otimes n}$ to $\mathbb{C}^2\otimes\mathbb{C}^2$ with $\Tr(\Lambda(\rho^{\otimes n}))>0$, the normalized output $\tau = \Lambda(\rho^{\otimes n})/\Tr(\Lambda(\rho^{\otimes n}))$ satisfies
\[
  \Tr(\Omega_2\,\tau) \le \frac{1+\sqrt{2}}{4}.
\]
\end{theorem}
\begin{proof}
\leanok
Combine Lemmas~\ref{lem:maps}, \ref{lem:certificate}, and~\ref{lem:choi-cert} with Theorem~\ref{thm:overlap-criterion}.

\begin{theorem}[Zero finite-round LOCC distillation rate of $\rho$]
\label{thm:zero-rate}
\lean{NPTEnt.Main.npt_zero_finiteRoundLOCC_distillableEntanglement}
\leanok
\uses{lem:state,lem:nonseparable,thm:cap,lem:locc,lem:zero}
The Werner state $\rho=(23I-3F)/1448$ on $\mathbb{C}^8\otimes\mathbb{C}^8$ is NPT, entangled, and has $E_D(\rho)=0$ under the finite-round, finite-outcome LOCC class of Lemma~\ref{lem:locc}.
\end{theorem}
\begin{proof}
\leanok
Since $(1+\sqrt{2})/4 < 1$, combining Lemmas~\ref{lem:state}, \ref{lem:nonseparable}, \ref{lem:locc}, and~\ref{lem:zero} with Theorem~\ref{thm:cap} gives $E_D(\rho)=0$.
\end{proof}

Separately from the finite-round LOCC rate theorem above, the uniform cap of Theorem~\ref{thm:cap} combines with Theorem~\ref{thm:cap-undistillable} to establish the standard Dür--Cirac--Lewenstein--Bru{\ss} finite-copy undistillability criterion for $\rho$.

\begin{theorem}[Finite-copy undistillability of $\rho$]
\label{thm:bound}
\lean{NPTEnt.Main.isFiniteCopyUndistillable_rho, NPTEnt.Main.partialTransposeFst_bipartiteCopies_rho_schmidtRankLE_two_nonneg, NPTEnt.Main.partialTransposeBob_bipartiteCopies_rho_schmidtRankLE_two_nonneg, NPTEnt.Main.npt_finiteCopyUndistillable}
\leanok
\uses{lem:state,lem:nonseparable,thm:cap,thm:cap-undistillable}
The Werner state $\rho=(23I-3F)/1448$ on $\mathbb{C}^8\otimes\mathbb{C}^8$ is an NPT entangled state that is finite-copy undistillable: for every $N\ge 1$ and every vector $\ket{\psi}\in(\mathbb{C}^8)^{\otimes N}\otimes(\mathbb{C}^8)^{\otimes N}$ of Schmidt rank at most $2$,
\[
  \operatorname{Re}\bra{\psi}(\rho^{\otimes N})^{\Gamma_A}\ket{\psi} \ge 0
  \qquad\text{and}\qquad
  \operatorname{Re}\bra{\psi}(\rho^{\otimes N})^{\Gamma_B}\ket{\psi} \ge 0.
\]
\end{theorem}
\begin{proof}
\leanok
By Lemmas~\ref{lem:state} and~\ref{lem:nonseparable}, $\rho$ is a positive semidefinite, trace-one, NPT entangled state. Applying Theorem~\ref{thm:cap-undistillable} with the uniform Bell-overlap cap $c = (1+\sqrt{2})/4 < 1$ from Theorem~\ref{thm:cap} yields finite-copy undistillability in the Alice partial-transpose convention. Because $\rho = (23I-3F)/1448$ is real symmetric, we have $\rho^{\Gamma_B} = \rho^{\Gamma_A}$ and hence $(\rho^{\otimes N})^{\Gamma_B} = (\rho^{\otimes N})^{\Gamma_A}$ entrywise, so the same inequality holds for the Bob partial transpose.
\end{proof}
